def _match_signature(expression: str, signatures: dict[str, dict[str, Any]]) -> str | None:
    ops = expression_ops(expression)
    feats = expression_features(expression)
    tokens = ops | feats
    for sig_name, info in signatures.items():
        kws = info["keywords"]
        hits = sum(1 for kw in kws if any(kw == t or kw in t for t in tokens))
        if hits >= min(2, len(kws)):
            return sig_name
    return None
