    def _hybrid_score(
        self,
        entry: dict[str, Any],

    ) -> float:

        entry_ops = expression_ops(entry.get("expression") or "")

            len(query_ops & entry_ops) / len(query_ops | entry_ops) if (query_ops | entry_ops) else 0.0
        )
        verdict_w = VERDICT_WEIGHT.get(str(entry.get("verdict")), 0.2)

            age_days = 30.0
        recency = math.exp(-age_days / 90.0)
